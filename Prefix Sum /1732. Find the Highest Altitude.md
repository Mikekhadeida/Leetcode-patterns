class Solution {
    public int largestAltitude(int[] gain) {
    int nums=0; 
    int maxNums =0; 

    int sum =0;

    for(int i =0; i<gain.length; i++){
    sum=sum+gain[i];

    maxNums=Math.max(maxNums, sum);    
    }
     return maxNums;

    }
}


/*
Time Complexity: O(n)
The algorithm runs in O(n) time because it goes through each
element in the input array once and updates the current altitude.

Space Complexity: O(1)
The algorithm uses O(1) extra space because only a few variables
are used and no additional data structures are needed.
*/
