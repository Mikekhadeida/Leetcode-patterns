class Solution {
    
    private boolean isVowel(char vowels){
       return vowels =='a' || vowels== 'e' || vowels == 'i' || vowels =='o' || vowels =='u';
    }

   public int maxVowels(String s, int k) {
    // char vowels = 'a', 'e', 'i', 'o', 'u';
    int count = 0;
    int max =0;
    // read first 3 letters to count if there is any letter, then create second for loop for the rest of letters 
    for(int i =0; i<k; i++){
        if(isVowel(s.charAt(i))){
            count++;
        }
    }
    max = count;
    // create another loop for second letter to rood as a slider;
    for(int i=k; i<s.length(); i++){
        if(isVowel(s.charAt(i - k))){
            count--;
        }
        if (isVowel(s.charAt(i))){
            count++;
        }
            max=Math.max(max,count);
    }

    return max;
   }
}

/* 
Time complexity: O(n) 
- Because each characters visit each letter only once. 

Space Complexity: O(1)
- Because we only use use a few variables
- (count & Max) No extra data structere is used. 
*/
